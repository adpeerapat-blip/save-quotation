// --- Main GET Handler ---
function doGet(e) {
  const action = e.parameter.action;
  let response;

  try {
    switch (action) {
      case "getDocuments":
        response = getDocuments();
        break;
      case "getQuotationDetails":
        response = getQuotationDetails(e.parameter.docNumber);
        break;
      case "getLogo": 
        response = getLogo();
        break;
      case "getPendingQuotationDetails":
        response = getPendingQuotationDetails();
        break;
      case "getAllUsers":
        response = getAllUsers();
        break;
      case "getUserSignature":
        response = getUserSignature(e.parameter.userIdentifier);
        break;
      // ฟังก์ชันใหม่: โหลดรายชื่อผู้อนุมัติ
      case "getApprovers":
        response = getApprovers();
        break;
      default:
        response = { success: false, message: "Invalid GET action." };
    }
  } catch (err) {
      response = { success: false, message: `Server error in GET: ${err.message}` };
      Logger.log(`GET Error: ${err.stack}`);
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- Main POST Handler ---
function doPost(e) {
  let response;
  try {
    const request = JSON.parse(e.postData.contents);
    const action = request.action;

    switch (action) {
      case "generateDocNumber":
        response = generateDocNumber();
        break;
      case "createQuotation":
        response = createQuotation(request.data);
        break;
      case "updateQuotation":
        response = updateQuotation(request.data);
        break;
      case "updateQuotationStatus":
        response = updateQuotationStatus(request.data);
        break;
      case "uploadLogo":
        response = uploadLogo(request.data);
        break;
      case "saveUserSignature":
        response = saveUserSignature(request.data);
        break;
      // ฟังก์ชันใหม่: เซฟลายเซ็นผู้อนุมัติลงชีต
      case "saveApproverSignature":
        response = saveApproverSignature(request.data);
        break;
      default:
        response = { success: false, message: "Invalid POST action." };
    }
  } catch (err) {
    response = { success: false, message: `Server error in POST: ${err.message}` };
    Logger.log(`POST Error: ${err.stack}`);
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- NEW Approvers Functions ---

function getApprovers() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Approvers");
    if (!sheet || sheet.getLastRow() < 2) return { success: true, data: [] };
    
    const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).getValues();
    // แก้ไข: ปรับเป็นตัวพิมพ์เล็กทั้งหมดเพื่อป้องกันปัญหา Case Sensitive (Name vs name)
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    
    const emailIdx = headers.indexOf("email");
    const nameIdx = headers.indexOf("name");
    const sigIdx = headers.indexOf("signatureurl");
    
    const approvers = data.map(row => ({
      email: emailIdx !== -1 ? row[emailIdx] : "",
      name: nameIdx !== -1 ? row[nameIdx] : "",
      signatureUrl: sigIdx !== -1 ? row[sigIdx] : ""
    })).filter(a => a.email); // คัดกรองบรรทัดว่างออก
    
    return { success: true, data: approvers };
  } catch (err) {
    Logger.log(`Error in getApprovers: ${err.stack}`);
    return { success: false, message: err.message };
  }
}

function saveApproverSignature(data) {
  const { email, name, signatureBase64 } = data; // รับค่า name มาด้วย
  if (!email || !signatureBase64) return { success: false, message: "Missing email or signature." };
  if (!signatureBase64.startsWith('data:image/png;base64,')) {
    return { success: false, message: "Invalid Base64 data." };
  }
  
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Approvers");
    if (!sheet) return { success: false, message: "Approvers sheet not found." };
    
    const sheetData = sheet.getDataRange().getValues();
    const headers = sheetData[0].map(h => String(h).trim().toLowerCase());
    const emailIdx = headers.indexOf("email");
    const sigIdx = headers.indexOf("signatureurl");
    const nameIdx = headers.indexOf("name");
    
    let rowIndex = -1;
    for (let i = 1; i < sheetData.length; i++) {
      if (emailIdx !== -1 && sheetData[i][emailIdx] === email) {
        rowIndex = i;
        break;
      }
    }
    
    // สร้างลิงก์ lh3
    const signatureUrl = saveSignatureToDrive(signatureBase64, `approver_${email.split('@')[0]}_${new Date().getTime()}`);
    
    if (rowIndex !== -1 && sigIdx !== -1) {
      // อัปเดตลิงก์ลายเซ็นเดิม
      sheet.getRange(rowIndex + 1, sigIdx + 1).setValue(signatureUrl);
      // ถ้าระบบส่งชื่อมาด้วย ให้อัปเดตชื่อลงไปป้องกันชื่อหาย
      if (nameIdx !== -1 && name) {
        sheet.getRange(rowIndex + 1, nameIdx + 1).setValue(name);
      }
    } else {
      // ถ้าไม่เจอในชีตก็สร้างใหม่ พร้อมบันทึกชื่อ
      sheet.appendRow([email, name || "", signatureUrl]); 
    }
    return { success: true, signatureUrl: signatureUrl };
  } catch (err) {
    Logger.log(`Error in saveApproverSignature: ${err.stack}`);
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}


// --- User Signature Functions ---

function saveUserSignature(data) {
  const { userIdentifier, signatureBase64 } = data; 
  if (!userIdentifier || !signatureBase64) {
    return { success: false, message: "Missing user identifier or signature." };
  }
  
  if (!signatureBase64.startsWith('data:image/png;base64,')) {
    return { success: false, message: "Invalid Base64 data. Must start with data:image/png;base64," };
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(10000); 

  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Users");
    if (!sheet) {
      return { success: false, message: "Users sheet not found. Please run setup." };
    }

    const dataRange = sheet.getDataRange();
    const values = dataRange.getValues();
    const headers = values[0].map(h => String(h).trim().toLowerCase());
    const idColIndex = headers.indexOf("useridentifier");
    const sigColIndex = headers.indexOf("signatureurl"); 

    let rowIndex = -1;
    for (let i = 1; i < values.length; i++) {
      if (idColIndex !== -1 && values[i][idColIndex] === userIdentifier) {
        rowIndex = i;
        break;
      }
    }

    const signatureUrl = saveSignatureToDrive(signatureBase64, `user_${userIdentifier}_${new Date().getTime()}`); 

    if (rowIndex !== -1 && sigColIndex !== -1) {
      sheet.getRange(rowIndex + 1, sigColIndex + 1).setValue(signatureUrl); 
    } else {
      sheet.appendRow([userIdentifier, signatureUrl]); 
    }

    return { success: true, message: "Signature saved.", signatureUrl: signatureUrl }; 

  } catch (err) { 
    Logger.log(`Error in saveUserSignature: ${err.stack}`);
    return { success: false, message: `Error saving signature: ${err.message}` };
  } finally {
    lock.releaseLock();
  }
}

function getUserSignature(userIdentifier) {
  if (!userIdentifier) {
    return { success: false, message: "Missing user identifier." };
  }

  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Users");
    if (!sheet || sheet.getLastRow() < 2) {
      return { success: true, signature: null }; 
    }

    const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).getValues();
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    const idColIndex = headers.indexOf("useridentifier");
    const sigColIndex = headers.indexOf("signatureurl"); 

    for (const row of data) {
      if (idColIndex !== -1 && row[idColIndex] === userIdentifier) {
        return { success: true, signature: sigColIndex !== -1 ? row[sigColIndex] : null }; 
      }
    }

    return { success: true, signature: null }; 

  } catch (err) {
    Logger.log(`Error in getUserSignature: ${err.stack}`);
    return { success: false, message: `Error getting signature: ${err.message}` };
  }
}

// --- Logo Functions ---
function uploadLogo(base64Data) {
  if (!base64Data) {
    return { success: false, message: "No logo data provided." };
  }
  try {
    const properties = PropertiesService.getScriptProperties();
    const oldFileId = properties.getProperty('logoFileId');

    if (oldFileId) {
      try {
        DriveApp.getFileById(oldFileId).setTrashed(true);
      } catch (e) {
        Logger.log(`Could not trash old logo file with ID ${oldFileId}. It might have been deleted already. Error: ${e.message}`);
      }
    }
    
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const decoded = Utilities.base64Decode(base64Data.split(',')[1]);
    const blob = Utilities.newBlob(decoded, MimeType.PNG, "company_logo.png");
    const file = folder.createFile(blob);

    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    const newFileId = file.getId();
    const newUrl = 'https://lh3.googleusercontent.com/d/' + newFileId;

    properties.setProperties({
      'logoFileId': newFileId,
      'logoUrl': newUrl
    });

    return { success: true, url: newUrl };
  } catch (err) {
    Logger.log(`Error in uploadLogo: ${err.stack}`);
    return { success: false, message: `Error uploading logo: ${err.message}` };
  }
}

// --- Action Functions ---
function updateQuotation(data) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const docNumber = data.docNumber.toString().replace(/'/g, "");

    // === 1. Update Quotations Sheet ===
    const qSheet = ss.getSheetByName("Quotations");
    const qHeaders = qSheet.getRange(1, 1, 1, qSheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    const qDocNumberColIndex = qHeaders.indexOf("docnumber");

    if (qDocNumberColIndex === -1) return { success: false, message: "Column docNumber not found in Quotations sheet." };

    const allDocNumbers = qSheet.getRange(2, qDocNumberColIndex + 1, qSheet.getLastRow() - 1, 1).getValues();
    const rowIndex = allDocNumbers.findIndex(row => row[0].toString().replace(/'/g, "") === docNumber);

    if (rowIndex === -1) {
      return { success: false, message: `Document ${docNumber} not found.` };
    }
    const rowToUpdate = rowIndex + 2;

    let signatureUrl;
    if (data.signature && data.signature.startsWith('data:image/png;base64,')) {
        signatureUrl = saveSignatureToDrive(data.signature, docNumber + "_edited");
    } else {
        signatureUrl = data.signature || "";
    }

    const columnMap = {
      requester: data.requester, // Fixed: ใช้ data.requester ให้ตรงกับฝั่ง Frontend
      status: "pending", 
      contactname: data.contactName,
      contactphone: `'${data.contactPhone}`,
      companyname: data.companyName,
      address: data.address,
      deliverydate: data.deliveryDate ? new Date(data.deliveryDate) : null,
      expirydate: data.expiryDate ? new Date(data.expiryDate) : null,
      creditdays: data.creditDays,
      machinemodel: data.machineModel,
      technician: data.technician,
      proposerphone: `'${data.proposerPhone}`,
      grandtotal: data.grandTotal,
      signatureurl: signatureUrl,
      remarks: data.remarks || "",
      approvername: "",
      approversignatureurl: "",
      rejectionreason: "",
      approvaldate: ""
    };
    
    qHeaders.forEach((header, index) => {
      if (columnMap.hasOwnProperty(header)) {
        qSheet.getRange(rowToUpdate, index + 1).setValue(columnMap[header]);
      }
    });

    // === 2. Update Items Sheet ===
    const iSheet = ss.getSheetByName("Items");
    const iData = iSheet.getDataRange().getValues();
    const iHeaders = iData[0].map(h => String(h).trim().toLowerCase());
    const iDocNumColIndex = iHeaders.indexOf("docnumber");

    if (iDocNumColIndex !== -1) {
      for (let i = iData.length - 1; i > 0; i--) {
        if (iData[i][iDocNumColIndex].toString().replace(/'/g, "") === docNumber) {
          iSheet.deleteRow(i + 1);
        }
      }
    }
    
    (data.items || []).forEach(item => {
      iSheet.appendRow([
        `'${docNumber}`, `'${item.code}`, item.name, item.quantity, item.unitPrice, item.discount, item.netPrice
      ]);
    });
    
    return { success: true, message: "Quotation updated successfully." };
  } catch (err) {
    Logger.log(`Error in updateQuotation: ${err.stack}`);
    return { success: false, message: `Error updating quotation: ${err.message}` };
  }
}

function getPendingQuotationDetails() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    const iSheet = ss.getSheetByName("Items");
    const iData = iSheet.getLastRow() > 1 ? iSheet.getRange(2, 1, iSheet.getLastRow() - 1, iSheet.getLastColumn()).getValues() : [];
    const iHeadersOriginal = iSheet.getRange(1, 1, 1, iSheet.getLastColumn()).getValues()[0];
    const iHeadersLower = iHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const iDocNumIndex = iHeadersLower.indexOf("docnumber");
    
    const itemsByDocNum = iData.reduce((acc, r) => {
      if (iDocNumIndex === -1) return acc;
      const docNum = r[iDocNumIndex].toString().replace(/'/g, "");
      if (!acc[docNum]) acc[docNum] = [];
      const item = iHeadersOriginal.reduce((obj, header, i) => {
        obj[header] = r[i];
        return obj;
      }, {});
      acc[docNum].push(item);
      return acc;
    }, {});


    const qSheet = ss.getSheetByName("Quotations");
    const qData = qSheet.getLastRow() > 1 ? qSheet.getRange(2, 1, qSheet.getLastRow() - 1, qSheet.getLastColumn()).getValues() : [];
    const qHeadersOriginal = qSheet.getRange(1, 1, 1, qSheet.getLastColumn()).getValues()[0];
    const qHeadersLower = qHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const qStatusIndex = qHeadersLower.indexOf("status");
    
    const dateIndices = {
        date: qHeadersLower.indexOf("date"),
        deliveryDate: qHeadersLower.indexOf("deliverydate"),
        expiryDate: qHeadersLower.indexOf("expirydate"),
        approvalDate: qHeadersLower.indexOf("approvaldate")
    };
    
    const pendingQuotations = qData
      .filter(r => qStatusIndex !== -1 && r[qStatusIndex] === 'pending')
      .map(r => {
        const quotation = qHeadersOriginal.reduce((obj, header, i) => {
           const isDateColumn = Object.values(dateIndices).includes(i);
            if (isDateColumn && r[i] instanceof Date) {
               obj[header] = r[i].toISOString();
            } else {
               obj[header] = r[i];
            }
          return obj;
        }, {});
        
        const docNum = quotation.docNumber ? quotation.docNumber.toString().replace(/'/g, "") : "";
        quotation.items = itemsByDocNum[docNum] || [];
        return quotation;
      });
      
    return { success: true, data: pendingQuotations };
  } catch(err) {
     Logger.log(`Error in getPendingQuotationDetails: ${err.stack}`);
    return { success: false, message: `Error fetching pending details: ${err.message}` };
  }
}

function getDocuments() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const quotationsSheet = ss.getSheetByName("Quotations");
    if (!quotationsSheet || quotationsSheet.getLastRow() < 2) {
      return { success: true, data: { all: [], pendingCount: 0 } };
    }

    const data = quotationsSheet.getRange(2, 1, quotationsSheet.getLastRow() - 1, quotationsSheet.getLastColumn()).getValues();
    const headers = quotationsSheet.getRange(1, 1, 1, quotationsSheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    
    const docNumberIndex = headers.indexOf("docnumber");
    const dateIndex = headers.indexOf("date");
    const requesterIndex = headers.indexOf("requester");
    const statusIndex = headers.indexOf("status");
    const approverNameIndex = headers.indexOf("approvername");
    const rejectionReasonIndex = headers.indexOf("rejectionreason");
    const contactNameIndex = headers.indexOf("contactname"); 
    const companyNameIndex = headers.indexOf("companyname"); 
    
    const documents = data.map(row => ({
      docNumber: docNumberIndex !== -1 ? row[docNumberIndex].toString().replace(/'/g, "") : "",
      date: (dateIndex !== -1 && row[dateIndex] instanceof Date) ? row[dateIndex].toISOString() : new Date().toISOString(),
      requester: requesterIndex !== -1 ? row[requesterIndex] : "",
      status: statusIndex !== -1 ? row[statusIndex] : "",
      approverName: approverNameIndex !== -1 ? (row[approverNameIndex] || '') : '',
      rejectionReason: rejectionReasonIndex !== -1 ? (row[rejectionReasonIndex] || '') : '',
      contactName: contactNameIndex !== -1 ? (row[contactNameIndex] || '') : '', 
      companyName: companyNameIndex !== -1 ? (row[companyNameIndex] || '') : ''  
    }));

    const pendingCount = documents.filter(doc => doc.status === 'pending').length;

    return { success: true, data: { all: documents, pendingCount: pendingCount } };
  } catch (err) {
    Logger.log(`Error in getDocuments: ${err.stack}`);
    return { success: false, message: `Error fetching documents: ${err.message}` };
  }
}

function getQuotationDetails(docNumber) {
  try {
    if (!docNumber) return { success: false, message: "Document number is required." };
    docNumber = docNumber.toString().replace(/'/g, "");
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    const qSheet = ss.getSheetByName("Quotations");
    const qData = qSheet.getDataRange().getValues();
    const qHeadersOriginal = qData.shift();
    const qHeadersLower = qHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const qDocNumberIndex = qHeadersLower.indexOf("docnumber");
    
    if (qDocNumberIndex === -1) return { success: false, message: "Column docNumber not found." };
    
    const qRow = qData.find(r => r[qDocNumberIndex].toString().replace(/'/g, "") === docNumber);

    if (!qRow) return { success: false, message: `Quotation ${docNumber} not found.` };
    
    const dateIndices = {
        date: qHeadersLower.indexOf("date"),
        deliveryDate: qHeadersLower.indexOf("deliverydate"),
        expiryDate: qHeadersLower.indexOf("expirydate"),
        approvalDate: qHeadersLower.indexOf("approvaldate")
    };
    
    const quotation = qHeadersOriginal.reduce((obj, header, i) => {
      const isDateColumn = Object.values(dateIndices).includes(i);
      if (isDateColumn && qRow[i] instanceof Date) {
        obj[header] = qRow[i].toISOString();
      } else {
        obj[header] = qRow[i];
      }
      return obj;
    }, {});

    const iSheet = ss.getSheetByName("Items");
    const iData = iSheet.getLastRow() > 1 ? iSheet.getRange(2, 1, iSheet.getLastRow() - 1, iSheet.getLastColumn()).getValues() : [];
    const iHeadersOriginal = iSheet.getRange(1, 1, 1, iSheet.getLastColumn()).getValues()[0];
    const iHeadersLower = iHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const docNumIndexItems = iHeadersLower.indexOf("docnumber");
    
    const items = iData
      .filter(r => docNumIndexItems !== -1 && r[docNumIndexItems].toString().replace(/'/g, "") === docNumber)
      .map(r => iHeadersOriginal.reduce((obj, header, i) => {
        obj[header] = r[i];
        return obj;
      }, {}));
      
    return { success: true, data: { ...quotation, items } };
  } catch(err) {
     Logger.log(`Error in getQuotationDetails: ${err.stack}`);
    return { success: false, message: `Error fetching details: ${err.message}` };
  }
}

function updateQuotationStatus(data) {
  try {
    const { docNumber, status, reason, approverName, signature } = data;
    if (!docNumber || !status) {
      return { success: false, message: "Missing required data for status update." };
    }
    
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Quotations");
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    const docNumberColIndex = headers.indexOf("docnumber") + 1;

    if (docNumberColIndex === 0) return { success: false, message: "Column docNumber not found." };

    const allDocNumbers = sheet.getRange(2, docNumberColIndex, sheet.getLastRow() - 1, 1).getValues();
    const rowIndex = allDocNumbers.findIndex(row => row[0].toString().replace(/'/g, "") === docNumber);
    
    if (rowIndex === -1) {
      return { success: false, message: `Document ${docNumber} not found for update.` };
    }
    
    const rowToUpdate = rowIndex + 2;
    
    sheet.getRange(rowToUpdate, headers.indexOf("status") + 1).setValue(status);
    const rejectIdx = headers.indexOf("rejectionreason");
    if (rejectIdx !== -1) sheet.getRange(rowToUpdate, rejectIdx + 1).setValue(reason || ""); 
    
    const apprNameIdx = headers.indexOf("approvername");
    const apprSigIdx = headers.indexOf("approversignatureurl");
    const appDateIdx = headers.indexOf("approvaldate");

    if (status === 'approved') {
      let signatureUrl;
      if (signature && signature.startsWith('data:image/png;base64,')) {
          signatureUrl = saveSignatureToDrive(signature, `${docNumber}_approver`);
      } else {
          signatureUrl = signature || "";
      }

      if (apprNameIdx !== -1) sheet.getRange(rowToUpdate, apprNameIdx + 1).setValue(approverName);
      if (apprSigIdx !== -1) sheet.getRange(rowToUpdate, apprSigIdx + 1).setValue(signatureUrl);
      if (appDateIdx !== -1) sheet.getRange(rowToUpdate, appDateIdx + 1).setValue(new Date());
    } else {
      if (apprNameIdx !== -1) sheet.getRange(rowToUpdate, apprNameIdx + 1).setValue("");
      if (apprSigIdx !== -1) sheet.getRange(rowToUpdate, apprSigIdx + 1).setValue("");
      if (appDateIdx !== -1) sheet.getRange(rowToUpdate, appDateIdx + 1).setValue("");
    }
    
    return { success: true, message: `Status updated to ${status}` };
  } catch (err) {
    Logger.log(`Error in updateQuotationStatus: ${err.stack}`);
    return { success: false, message: `Error updating status: ${err.message}` };
  }
}

function ensureSheetWithHeaders(spreadsheet, sheetName, headers) {
  let sheet = spreadsheet.getSheetByName(sheetName);
  if (!sheet) {
    sheet = spreadsheet.insertSheet(sheetName);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
  }
  return sheet;
}

function createQuotation(data) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    const quotationHeaders = [
      "docNumber", "date", "requester", "status", "contactName", "contactPhone", "companyName", "address", 
      "deliveryDate", "expiryDate", "creditDays", "machineModel", "technician", 
      "proposerPhone", "grandTotal", "signatureUrl", "approverName", "approverSignatureUrl",
      "rejectionReason", "remarks", "approvalDate"
    ];
    const quotationsSheet = ensureSheetWithHeaders(ss, "Quotations", quotationHeaders);

    const itemHeaders = ["docNumber", "code", "name", "quantity", "unitPrice", "discount", "netPrice"];
    const itemsSheet = ensureSheetWithHeaders(ss, "Items", itemHeaders);

    let signatureUrl;
    if (data.signature && data.signature.startsWith('data:image/png;base64,')) {
        signatureUrl = saveSignatureToDrive(data.signature, data.docNumber);
    } else {
        signatureUrl = data.signature || "";
    }
    
    quotationsSheet.appendRow([
      `'${data.docNumber}`, new Date(), data.requester, "pending", data.contactName, `'${data.contactPhone}`, data.companyName, data.address,
      data.deliveryDate ? new Date(data.deliveryDate) : null, 
      data.expiryDate ? new Date(data.expiryDate) : null, 
      data.creditDays, data.machineModel, data.technician,
      `'${data.proposerPhone}`, data.grandTotal, signatureUrl, "", "", "", data.remarks || "", ""
    ]);

    (data.items || []).forEach(item => {
      itemsSheet.appendRow([
        `'${data.docNumber}`, `'${item.code}`, item.name, item.quantity, item.unitPrice, item.discount, item.netPrice
      ]);
    });
    
    return { success: true, message: "Quotation created successfully." };
  } catch (err) {
    Logger.log(`Error in createQuotation: ${err.stack}`);
    return { success: false, message: `Error creating quotation: ${err.message}` };
  }
}

function generateDocNumber() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Quotations");
    const lock = LockService.getScriptLock();
    lock.waitLock(30000); 

    let newDocNumber;
    try {
      const now = new Date();
      const year = now.getFullYear();
      const month = (now.getMonth() + 1).toString().padStart(2, '0');
      const prefix = `QT-${year}${month}-`;

      if (!sheet || sheet.getLastRow() < 2) {
        newDocNumber = `${prefix}001`;
      } else {
        const docNumbers = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues()
                              .flat()
                              .map(d => d.toString().replace(/'/g, ""));
        
        const lastNumberInMonth = docNumbers
          .filter(docNum => docNum.startsWith(prefix))
          .map(docNum => parseInt(docNum.split('-')[2], 10))
          .reduce((max, current) => current > max ? current : max, 0);

        const newNumber = (lastNumberInMonth + 1).toString().padStart(3, '0');
        newDocNumber = `${prefix}${newNumber}`;
      }
    } finally {
      lock.releaseLock();
    }
    
    return { success: true, docNumber: newDocNumber };

  } catch (err) {
    Logger.log(`Error in generateDocNumber: ${err.stack}`);
    return { success: false, message: "Could not generate document number." };
  }
}

function saveSignatureToDrive(base64Data, fileName) {
  try {
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const decoded = Utilities.base64Decode(base64Data.split(',')[1]);
    const blob = Utilities.newBlob(decoded, MimeType.PNG, `Signature_${fileName}.png`);
    const file = folder.createFile(blob);
    
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    return 'https://lh3.googleusercontent.com/d/' + file.getId();
    
  } catch (err) {
    Logger.log(`Error saving signature: ${err.stack}`);
    return `Error saving signature: ${err.message}`;
  }
}
