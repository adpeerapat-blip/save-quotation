// --- Main GET Handler ---
function doGet(e) {
  const action = e.parameter.action;
  let response;

  try {
    switch (action) {
      case "getDocuments":
        response = getDocuments();
        break;
      case "getQuotationDetails":
        response = getQuotationDetails(e.parameter.docNumber);
        break;
      case "getLogo": 
        response = getLogo();
        break;
      case "getPendingQuotationDetails":
        response = getPendingQuotationDetails();
        break;
      case "getAllUsers":
        response = getAllUsers();
        break;
      case "getUserSignature":
        response = getUserSignature(e.parameter.userIdentifier);
        break;
      // ฟังก์ชันใหม่: โหลดรายชื่อผู้อนุมัติ
      case "getApprovers":
        response = getApprovers();
        break;
      // ฟังก์ชันโอนย้ายข้อมูลทั้งหมดไป Firebase
      case "getAllQuotationsFull":
        response = getAllQuotationsFull();
        break;
      case "syncFromFirebase":
        response = { success: false, message: "syncFromFirebase ต้องใช้คำขอแบบ POST เท่านั้น" };
        break;
      default:
        response = { success: false, message: "Invalid GET action." };
    }
  } catch (err) {
      response = { success: false, message: `Server error in GET: ${err.message}` };
      Logger.log(`GET Error: ${err.stack}`);
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- Main POST Handler ---
function doPost(e) {
  let response;
  try {
    const request = JSON.parse(e.postData.contents);
    const action = request.action;

    switch (action) {
      case "generateDocNumber":
        response = generateDocNumber();
        break;
      case "createQuotation":
        response = createQuotation(request.data);
        break;
      case "updateQuotation":
        response = updateQuotation(request.data);
        break;
      case "updateQuotationStatus":
        response = updateQuotationStatus(request.data);
        break;
      case "uploadLogo":
        response = uploadLogo(request.data);
        break;
      case "saveUserSignature":
        response = saveUserSignature(request.data);
        break;
      // ฟังก์ชันใหม่: เซฟลายเซ็นผู้อนุมัติลงชีต
      case "saveApproverSignature":
        response = saveApproverSignature(request.data);
        break;
      case "saveApprover":
        response = saveApprover(request.data);
        break;
      case "deleteApprover":
        response = deleteApprover(request.data);
        break;
      case "syncFromFirebase":
        response = syncFromFirebase(request.data);
        break;
      default:
        response = { success: false, message: "Invalid POST action." };
    }
  } catch (err) {
    response = { success: false, message: `Server error in POST: ${err.message}` };
    Logger.log(`POST Error: ${err.stack}`);
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- NEW Approvers Functions ---

function getApprovers() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Approvers");
    if (!sheet) return { success: true, data: [] };
    
    const allData = sheet.getDataRange().getValues();
    if (allData.length < 2) return { success: true, data: [] };
    
    const headers = allData[0].map(h => String(h).trim().toLowerCase());
    const data = allData.slice(1);
    
    const emailIdx = headers.indexOf("email");
    const nameIdx = headers.indexOf("name");
    const sigIdx = headers.indexOf("signatureurl");
    
    const approvers = data.map(row => ({
      email: emailIdx !== -1 ? row[emailIdx] : "",
      name: nameIdx !== -1 ? row[nameIdx] : "",
      signatureUrl: sigIdx !== -1 ? row[sigIdx] : ""
    })).filter(a => a.email); // คัดกรองบรรทัดว่างออก
    
    return { success: true, data: approvers };
  } catch (err) {
    Logger.log(`Error in getApprovers: ${err.stack}`);
    return { success: false, message: err.message };
  }
}

function saveApproverSignature(data) {
  const { email, name, signatureBase64 } = data; // รับค่า name มาด้วย
  if (!email || !signatureBase64) return { success: false, message: "Missing email or signature." };
  if (!signatureBase64.startsWith('data:image/png;base64,')) {
    return { success: false, message: "Invalid Base64 data." };
  }
  
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Approvers");
    if (!sheet) return { success: false, message: "Approvers sheet not found." };
    
    const sheetData = sheet.getDataRange().getValues();
    const headers = sheetData[0].map(h => String(h).trim().toLowerCase());
    const emailIdx = headers.indexOf("email");
    const sigIdx = headers.indexOf("signatureurl");
    const nameIdx = headers.indexOf("name");
    
    let rowIndex = -1;
    for (let i = 1; i < sheetData.length; i++) {
      if (emailIdx !== -1 && sheetData[i][emailIdx] === email) {
        rowIndex = i;
        break;
      }
    }
    
    // สร้างลิงก์ lh3
    const signatureUrl = saveSignatureToDrive(signatureBase64, `approver_${email.split('@')[0]}_${new Date().getTime()}`);
    
    if (rowIndex !== -1 && sigIdx !== -1) {
      // อัปเดตลิงก์ลายเซ็นเดิม
      sheet.getRange(rowIndex + 1, sigIdx + 1).setValue(signatureUrl);
      // ถ้าระบบส่งชื่อมาด้วย ให้อัปเดตชื่อลงไปป้องกันชื่อหาย
      if (nameIdx !== -1 && name) {
        sheet.getRange(rowIndex + 1, nameIdx + 1).setValue(name);
      }
    } else {
      // ถ้าไม่เจอในชีตก็สร้างใหม่ พร้อมบันทึกชื่อ
      sheet.appendRow([email, name || "", signatureUrl]); 
    }
    return { success: true, signatureUrl: signatureUrl };
  } catch (err) {
    Logger.log(`Error in saveApproverSignature: ${err.stack}`);
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}

function saveApprover(data) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const { email, name, signatureUrl } = data;
    if (!email) return { success: false, message: "Missing email." };
    const cleanEmail = email.trim().toLowerCase();
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Approvers");
    if (!sheet) return { success: false, message: "Approvers sheet not found." };
    
    const sheetData = sheet.getDataRange().getValues();
    const headers = sheetData[0].map(h => String(h).trim().toLowerCase());
    const emailIdx = headers.indexOf("email");
    const nameIdx = headers.indexOf("name");
    const sigIdx = headers.indexOf("signatureurl");
    
    let rowIndex = -1;
    for (let i = 1; i < sheetData.length; i++) {
      if (emailIdx !== -1 && String(sheetData[i][emailIdx]).trim().toLowerCase() === cleanEmail) {
        rowIndex = i;
        break;
      }
    }
    
    if (rowIndex !== -1) {
      if (nameIdx !== -1 && name !== undefined) {
        sheet.getRange(rowIndex + 1, nameIdx + 1).setValue(name);
      }
      if (sigIdx !== -1 && signatureUrl !== undefined) {
        sheet.getRange(rowIndex + 1, sigIdx + 1).setValue(signatureUrl);
      }
    } else {
      sheet.appendRow([cleanEmail, name || "", signatureUrl || ""]);
    }
    return { success: true, message: "Approver saved successfully." };
  } catch (err) {
    Logger.log(`Error in saveApprover: ${err.stack}`);
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}

function deleteApprover(data) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const { email } = data;
    if (!email) return { success: false, message: "Missing email." };
    const cleanEmail = email.trim().toLowerCase();
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Approvers");
    if (!sheet) return { success: false, message: "Approvers sheet not found." };
    
    const sheetData = sheet.getDataRange().getValues();
    const headers = sheetData[0].map(h => String(h).trim().toLowerCase());
    const emailIdx = headers.indexOf("email");
    if (emailIdx === -1) return { success: false, message: "Column email not found." };
    
    for (let i = sheetData.length - 1; i >= 1; i--) {
      if (String(sheetData[i][emailIdx]).trim().toLowerCase() === cleanEmail) {
        sheet.deleteRow(i + 1);
        return { success: true, message: "Approver deleted successfully." };
      }
    }
    return { success: true, message: "Approver not found or already deleted." };
  } catch (err) {
    Logger.log(`Error in deleteApprover: ${err.stack}`);
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}

function syncFromFirebase(payload) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const type = payload.type; // 'approvers', 'users', 'documents'
    const mode = payload.mode || 'upsert'; // 'upsert' or 'overwrite'
    const isFirstChunk = payload.isFirstChunk || false;

    // Helper ป้องกันข้อความยาวเกิน 50,000 ตัวอักษร (Google Sheets Cell Limit) และแปลง Base64 เป็น Google Drive URL
    const sigCache = {};
    function sanitizeForCell(val, nameForDrive) {
      if (!val) return '';
      const str = String(val).trim();
      if (str.startsWith('http://') || str.startsWith('https://')) return str;
      if (str.startsWith('data:image/')) {
        if (sigCache[str]) return sigCache[str];
        try {
          const driveUrl = saveSignatureToDrive(str, nameForDrive || ('sig_' + new Date().getTime()));
          if (driveUrl && driveUrl.startsWith('https://')) {
            sigCache[str] = driveUrl;
            return driveUrl;
          }
        } catch (e) {
          Logger.log('Error saving sig to drive: ' + e.message);
        }
        return '';
      }
      if (str.length > 45000) return str.substring(0, 45000);
      return str;
    }

    function ensureSheetCapacity(sheet, neededTotalRows) {
      const maxRows = sheet.getMaxRows();
      if (maxRows < neededTotalRows) {
        sheet.insertRowsAfter(maxRows, neededTotalRows - maxRows);
      }
    }

    function clearSheetDataBelowHeader(sheet) {
      const lastRow = sheet.getLastRow();
      const maxCols = sheet.getMaxColumns();
      if (lastRow > 1 && maxCols > 0) {
        sheet.getRange(2, 1, lastRow - 1, maxCols).clearContent();
      }
    }

    if (type === 'approvers') {
      const approverHeaders = ["email", "name", "signatureUrl"];
      const sheet = ensureSheetWithHeaders(ss, "Approvers", approverHeaders);
      const items = payload.data || [];
      if (mode === 'overwrite') {
        clearSheetDataBelowHeader(sheet);
        if (items.length > 0) {
          const rows = items.map(a => [
            a.email ? a.email.trim().toLowerCase() : '',
            sanitizeForCell(a.name || ''),
            sanitizeForCell(a.signatureUrl || '', 'approver_' + (a.email ? a.email.split('@')[0] : 'sig'))
          ]);
          ensureSheetCapacity(sheet, rows.length + 1);
          sheet.getRange(2, 1, rows.length, 3).setValues(rows);
        }
      } else {
        const existingData = sheet.getDataRange().getValues();
        const emailMap = {};
        for (let i = 1; i < existingData.length; i++) {
          const email = String(existingData[i][0]).trim().toLowerCase();
          if (email) emailMap[email] = i + 1;
        }
        items.forEach(a => {
          const email = a.email ? a.email.trim().toLowerCase() : '';
          if (!email) return;
          const nameVal = sanitizeForCell(a.name || '');
          const sigVal = sanitizeForCell(a.signatureUrl || '', 'approver_' + (a.email ? a.email.split('@')[0] : 'sig'));
          if (emailMap[email]) {
            sheet.getRange(emailMap[email], 1, 1, 3).setValues([[email, nameVal, sigVal]]);
          } else {
            sheet.appendRow([email, nameVal, sigVal]);
          }
        });
      }
      return { success: true, message: `Synced ${items.length} approvers.` };
    }

    if (type === 'users') {
      const userHeaders = ["userIdentifier", "signatureUrl"];
      const sheet = ensureSheetWithHeaders(ss, "Users", userHeaders);
      const items = payload.data || [];
      if (mode === 'overwrite') {
        clearSheetDataBelowHeader(sheet);
        if (items.length > 0) {
          const rows = items.map(u => [
            sanitizeForCell(u.userIdentifier || ''),
            sanitizeForCell(u.signatureUrl || '', 'user_' + (u.userIdentifier || 'sig'))
          ]);
          ensureSheetCapacity(sheet, rows.length + 1);
          sheet.getRange(2, 1, rows.length, 2).setValues(rows);
        }
      } else {
        const existingData = sheet.getDataRange().getValues();
        const userMap = {};
        for (let i = 1; i < existingData.length; i++) {
          const id = String(existingData[i][0]).trim();
          if (id) userMap[id] = i + 1;
        }
        items.forEach(u => {
          const id = (u.userIdentifier || '').trim();
          if (!id) return;
          const sigVal = sanitizeForCell(u.signatureUrl || '', 'user_' + id);
          if (userMap[id]) {
            sheet.getRange(userMap[id], 1, 1, 2).setValues([[id, sigVal]]);
          } else {
            sheet.appendRow([id, sigVal]);
          }
        });
      }
      return { success: true, message: `Synced ${items.length} users.` };
    }

    if (type === 'documents') {
      const quotationHeaders = [
        "docNumber", "date", "requester", "status", "contactName", "contactPhone", "companyName", "address", 
        "deliveryDate", "expiryDate", "creditDays", "machineModel", "technician", 
        "proposerPhone", "grandTotal", "signatureUrl", "approverName", "approverSignatureUrl", 
        "rejectionReason", "remarks", "approvalDate"
      ];
      const qSheet = ensureSheetWithHeaders(ss, "Quotations", quotationHeaders);

      const itemHeaders = ["docNumber", "code", "name", "quantity", "unitPrice", "discount", "netPrice"];
      const iSheet = ensureSheetWithHeaders(ss, "Items", itemHeaders);

      const docs = payload.data || [];

      if (mode === 'overwrite' && isFirstChunk) {
        clearSheetDataBelowHeader(qSheet);
        clearSheetDataBelowHeader(iSheet);
      }

      if (mode === 'overwrite') {
        const qRows = [];
        const iRows = [];

        docs.forEach(doc => {
          const docNum = String(doc.docNumber || '').replace(/'/g, '').trim();
          if (!docNum) return;
          qRows.push([
            `'${docNum}`,
            doc.date ? new Date(doc.date) : new Date(),
            sanitizeForCell(doc.requester || ''),
            doc.status || 'pending',
            sanitizeForCell(doc.contactName || ''),
            `'${sanitizeForCell(doc.contactPhone || '')}`,
            sanitizeForCell(doc.companyName || ''),
            sanitizeForCell(doc.address || ''),
            doc.deliveryDate ? new Date(doc.deliveryDate) : '',
            doc.expiryDate ? new Date(doc.expiryDate) : '',
            doc.creditDays || 0,
            sanitizeForCell(doc.machineModel || ''),
            sanitizeForCell(doc.technician || ''),
            `'${sanitizeForCell(doc.proposerPhone || '')}`,
            doc.grandTotal || '',
            sanitizeForCell(doc.signatureUrl || '', 'req_' + docNum),
            sanitizeForCell(doc.approverName || ''),
            sanitizeForCell(doc.approverSignatureUrl || '', 'appr_' + docNum),
            sanitizeForCell(doc.rejectionReason || ''),
            sanitizeForCell(doc.remarks || ''),
            doc.approvalDate ? new Date(doc.approvalDate) : ''
          ]);

          (doc.items || []).forEach(item => {
            iRows.push([
              `'${docNum}`,
              `'${sanitizeForCell(item.code || '')}`,
              sanitizeForCell(item.name || ''),
              item.quantity || 1,
              item.unitPrice || 0,
              item.discount || 0,
              item.netPrice || 0
            ]);
          });
        });

        if (qRows.length > 0) {
          const startQRow = qSheet.getLastRow() + 1;
          ensureSheetCapacity(qSheet, startQRow + qRows.length - 1);
          qSheet.getRange(startQRow, 1, qRows.length, quotationHeaders.length).setValues(qRows);
        }
        if (iRows.length > 0) {
          const startIRow = iSheet.getLastRow() + 1;
          ensureSheetCapacity(iSheet, startIRow + iRows.length - 1);
          iSheet.getRange(startIRow, 1, iRows.length, itemHeaders.length).setValues(iRows);
        }

      } else {
        // Upsert mode: ทำงานในหน่วยความจำ (in-memory) เพื่อความเร็วสูงสุด ไม่เรียก API ทีละแถว
        const qData = qSheet.getDataRange().getValues();
        const qDocMap = {};
        for (let i = 1; i < qData.length; i++) {
          const dNum = String(qData[i][0]).replace(/'/g, '').trim();
          if (dNum) qDocMap[dNum] = i; // เก็บ 0-based index ใน qData
        }

        const incomingDocNums = new Set(docs.map(d => String(d.docNumber || '').replace(/'/g, '').trim()));
        const newQRowsToAppend = [];
        const incomingIRows = [];

        docs.forEach(doc => {
          const docNum = String(doc.docNumber || '').replace(/'/g, '').trim();
          if (!docNum) return;
          const rowValues = [
            `'${docNum}`,
            doc.date ? new Date(doc.date) : new Date(),
            sanitizeForCell(doc.requester || ''),
            doc.status || 'pending',
            sanitizeForCell(doc.contactName || ''),
            `'${sanitizeForCell(doc.contactPhone || '')}`,
            sanitizeForCell(doc.companyName || ''),
            sanitizeForCell(doc.address || ''),
            doc.deliveryDate ? new Date(doc.deliveryDate) : '',
            doc.expiryDate ? new Date(doc.expiryDate) : '',
            doc.creditDays || 0,
            sanitizeForCell(doc.machineModel || ''),
            sanitizeForCell(doc.technician || ''),
            `'${sanitizeForCell(doc.proposerPhone || '')}`,
            doc.grandTotal || '',
            sanitizeForCell(doc.signatureUrl || '', 'req_' + docNum),
            sanitizeForCell(doc.approverName || ''),
            sanitizeForCell(doc.approverSignatureUrl || '', 'appr_' + docNum),
            sanitizeForCell(doc.rejectionReason || ''),
            sanitizeForCell(doc.remarks || ''),
            doc.approvalDate ? new Date(doc.approvalDate) : ''
          ];

          if (qDocMap[docNum] !== undefined) {
            qData[qDocMap[docNum]] = rowValues;
          } else {
            newQRowsToAppend.push(rowValues);
          }

          (doc.items || []).forEach(item => {
            incomingIRows.push([
              `'${docNum}`,
              `'${sanitizeForCell(item.code || '')}`,
              sanitizeForCell(item.name || ''),
              item.quantity || 1,
              item.unitPrice || 0,
              item.discount || 0,
              item.netPrice || 0
            ]);
          });
        });

        // 1. บันทึก Quotations: อัปเดตแถวเดิม + ต่อท้ายแถวใหม่
        if (qData.length > 1) {
          ensureSheetCapacity(qSheet, qData.length);
          qSheet.getRange(2, 1, qData.length - 1, quotationHeaders.length).setValues(qData.slice(1));
        }
        if (newQRowsToAppend.length > 0) {
          const startQRow = qSheet.getLastRow() + 1;
          ensureSheetCapacity(qSheet, startQRow + newQRowsToAppend.length - 1);
          qSheet.getRange(startQRow, 1, newQRowsToAppend.length, quotationHeaders.length).setValues(newQRowsToAppend);
        }

        // 2. จัดการ Items: รวมของเดิมที่ไม่ซ้ำกับ incomingDocNums + incomingIRows ใหม่
        const iData = iSheet.getDataRange().getValues();
        const existingItemsKept = iData.length > 1
          ? iData.slice(1).filter(r => !incomingDocNums.has(String(r[0]).replace(/'/g, '').trim()))
          : [];
        const finalItems = existingItemsKept.concat(incomingIRows);

        // ล้างข้อมูลเก่าด้านล่างหัวตารางโดยไม่ลบแถว (ป้องกัน error: ขออภัย ลบทุกแถวที่ไม่ได้ตรึงไว้ไม่ได้)
        clearSheetDataBelowHeader(iSheet);
        if (finalItems.length > 0) {
          ensureSheetCapacity(iSheet, finalItems.length + 1);
          iSheet.getRange(2, 1, finalItems.length, itemHeaders.length).setValues(finalItems);
        }
      }

      return { success: true, message: `Synced ${docs.length} documents.` };
    }

    return { success: false, message: `Unknown sync type: ${type}` };
  } catch (err) {
    Logger.log(`Error in syncFromFirebase: ${err.stack}`);
    return { success: false, message: err.message };
  } finally {
    lock.releaseLock();
  }
}


// --- User Signature Functions ---

function saveUserSignature(data) {
  const { userIdentifier, signatureBase64 } = data; 
  if (!userIdentifier || !signatureBase64) {
    return { success: false, message: "Missing user identifier or signature." };
  }
  
  if (!signatureBase64.startsWith('data:image/png;base64,')) {
    return { success: false, message: "Invalid Base64 data. Must start with data:image/png;base64," };
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(10000); 

  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Users");
    if (!sheet) {
      return { success: false, message: "Users sheet not found. Please run setup." };
    }

    const dataRange = sheet.getDataRange();
    const values = dataRange.getValues();
    const headers = values[0].map(h => String(h).trim().toLowerCase());
    const idColIndex = headers.indexOf("useridentifier");
    const sigColIndex = headers.indexOf("signatureurl"); 

    let rowIndex = -1;
    for (let i = 1; i < values.length; i++) {
      if (idColIndex !== -1 && values[i][idColIndex] === userIdentifier) {
        rowIndex = i;
        break;
      }
    }

    const signatureUrl = saveSignatureToDrive(signatureBase64, `user_${userIdentifier}_${new Date().getTime()}`); 

    if (rowIndex !== -1 && sigColIndex !== -1) {
      sheet.getRange(rowIndex + 1, sigColIndex + 1).setValue(signatureUrl); 
    } else {
      sheet.appendRow([userIdentifier, signatureUrl]); 
    }

    return { success: true, message: "Signature saved.", signatureUrl: signatureUrl };

function updateQuotation(data) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const docNumber = data.docNumber.toString().replace(/'/g, "");

    // === 1. Update Quotations Sheet ===
    const qSheet = ss.getSheetByName("Quotations");
    const qHeaders = qSheet.getRange(1, 1, 1, qSheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    const qDocNumberColIndex = qHeaders.indexOf("docnumber");

    if (qDocNumberColIndex === -1) return { success: false, message: "Column docNumber not found in Quotations sheet." };

    const allDocNumbers = qSheet.getRange(2, qDocNumberColIndex + 1, qSheet.getLastRow() - 1, 1).getValues();
    const rowIndex = allDocNumbers.findIndex(row => row[0].toString().replace(/'/g, "") === docNumber);

    if (rowIndex === -1) {
      return { success: false, message: `Document ${docNumber} not found.` };
    }
    const rowToUpdate = rowIndex + 2;

    let signatureUrl;
    if (data.signature && data.signature.startsWith('data:image/png;base64,')) {
        signatureUrl = saveSignatureToDrive(data.signature, docNumber + "_edited");
    } else {
        signatureUrl = data.signature || "";
    }

    const columnMap = {
      requester: data.requester, // Fixed: ใช้ data.requester ให้ตรงกับฝั่ง Frontend
      status: "pending", 
      contactname: data.contactName,
      contactphone: `'${data.contactPhone}`,
      companyname: data.companyName,
      address: data.address,
      deliverydate: data.deliveryDate ? new Date(data.deliveryDate) : null,
      expirydate: data.expiryDate ? new Date(data.expiryDate) : null,
      creditdays: data.creditDays,
      machinemodel: data.machineModel,
      technician: data.technician,
      proposerphone: `'${data.proposerPhone}`,
      grandtotal: data.grandTotal,
      signatureurl: signatureUrl,
      remarks: data.remarks || "",
      approvername: "",
      approversignatureurl: "",
      rejectionreason: "",
      approvaldate: ""
    };
    
    qHeaders.forEach((header, index) => {
      if (columnMap.hasOwnProperty(header)) {
        qSheet.getRange(rowToUpdate, index + 1).setValue(columnMap[header]);
      }
    });

    // === 2. Update Items Sheet ===
    const iSheet = ss.getSheetByName("Items");
    const iData = iSheet.getDataRange().getValues();
    const iHeaders = iData[0].map(h => String(h).trim().toLowerCase());
    const iDocNumColIndex = iHeaders.indexOf("docnumber");

    if (iDocNumColIndex !== -1) {
      for (let i = iData.length - 1; i > 0; i--) {
        if (iData[i][iDocNumColIndex].toString().replace(/'/g, "") === docNumber) {
          iSheet.deleteRow(i + 1);
        }
      }
    }
    
    (data.items || []).forEach(item => {
      iSheet.appendRow([
        `'${docNumber}`, `'${item.code}`, item.name, item.quantity, item.unitPrice, item.discount, item.netPrice
      ]);
    });
    
    return { success: true, message: "Quotation updated successfully." };
  } catch (err) {
    Logger.log(`Error in updateQuotation: ${err.stack}`);
    return { success: false, message: `Error updating quotation: ${err.message}` };
  }
}

function getPendingQuotationDetails() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    const iSheet = ss.getSheetByName("Items");
    const iData = iSheet.getLastRow() > 1 ? iSheet.getRange(2, 1, iSheet.getLastRow() - 1, iSheet.getLastColumn()).getValues() : [];
    const iHeadersOriginal = iSheet.getRange(1, 1, 1, iSheet.getLastColumn()).getValues()[0];
    const iHeadersLower = iHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const iDocNumIndex = iHeadersLower.indexOf("docnumber");
    
    const itemsByDocNum = iData.reduce((acc, r) => {
      if (iDocNumIndex === -1) return acc;
      const docNum = r[iDocNumIndex].toString().replace(/'/g, "");
      if (!acc[docNum]) acc[docNum] = [];
      const item = iHeadersOriginal.reduce((obj, header, i) => {
        obj[header] = r[i];
        return obj;
      }, {});
      acc[docNum].push(item);
      return acc;
    }, {});


    const qSheet = ss.getSheetByName("Quotations");
    const qData = qSheet.getLastRow() > 1 ? qSheet.getRange(2, 1, qSheet.getLastRow() - 1, qSheet.getLastColumn()).getValues() : [];
    const qHeadersOriginal = qSheet.getRange(1, 1, 1, qSheet.getLastColumn()).getValues()[0];
    const qHeadersLower = qHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const qStatusIndex = qHeadersLower.indexOf("status");
    
    const dateIndices = {
        date: qHeadersLower.indexOf("date"),
        deliveryDate: qHeadersLower.indexOf("deliverydate"),
        expiryDate: qHeadersLower.indexOf("expirydate"),
        approvalDate: qHeadersLower.indexOf("approvaldate")
    };
    
    const pendingQuotations = qData
      .filter(r => qStatusIndex !== -1 && r[qStatusIndex] === 'pending')
      .map(r => {
        const quotation = qHeadersOriginal.reduce((obj, header, i) => {
           const isDateColumn = Object.values(dateIndices).includes(i);
            if (isDateColumn && r[i] instanceof Date) {
               obj[header] = r[i].toISOString();
            } else {
               obj[header] = r[i];
            }
          return obj;
        }, {});
        
        const docNum = quotation.docNumber ? quotation.docNumber.toString().replace(/'/g, "") : "";
        quotation.items = itemsByDocNum[docNum] || [];
        return quotation;
      });
      
    return { success: true, data: pendingQuotations };
  } catch(err) {
     Logger.log(`Error in getPendingQuotationDetails: ${err.stack}`);
    return { success: false, message: `Error fetching pending details: ${err.message}` };
  }
}

function getDocuments() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const quotationsSheet = ss.getSheetByName("Quotations");
    if (!quotationsSheet) {
      return { success: true, data: { all: [], pendingCount: 0 } };
    }

    const allData = quotationsSheet.getDataRange().getValues();
    if (allData.length < 2) {
      return { success: true, data: { all: [], pendingCount: 0 } };
    }

    const headers = allData[0].map(h => String(h).trim().toLowerCase());
    const data = allData.slice(1);
    
    const defaultHeaders = [
      "docNumber", "date", "requester", "status", "contactName", "contactPhone", "companyName", "address", 
      "deliveryDate", "expiryDate", "creditDays", "machineModel", "technician", 
      "proposerPhone", "grandTotal", "signatureUrl", "approverName", "approverSignatureUrl", 
      "rejectionReason", "remarks", "approvalDate"
    ];

    // ซ่อมแซมหัวตารางแถวที่ 1 อัตโนมัติหากพบว่า docNumber หายไปหรือโดนพิมพ์ทับ
    if (headers.indexOf("docnumber") === -1) {
      try {
        quotationsSheet.getRange(1, 1, 1, defaultHeaders.length).setValues([defaultHeaders]);
      } catch (e) {
        Logger.log("Could not auto-repair headers: " + e.message);
      }
    }

    // กำหนด Index โดยมี Fallback ตำแหน่งคอลัมน์มาตรฐาน (A=0, B=1, ...) ป้องกันค่าว่างหาก header เพี้ยน
    const docNumberIndex = headers.indexOf("docnumber") !== -1 ? headers.indexOf("docnumber") : 0;
    const dateIndex = headers.indexOf("date") !== -1 ? headers.indexOf("date") : 1;
    const requesterIndex = headers.indexOf("requester") !== -1 ? headers.indexOf("requester") : 2;
    const statusIndex = headers.indexOf("status") !== -1 ? headers.indexOf("status") : 3;
    const contactNameIndex = headers.indexOf("contactname") !== -1 ? headers.indexOf("contactname") : 4;
    const companyNameIndex = headers.indexOf("companyname") !== -1 ? headers.indexOf("companyname") : 6;
    const approverNameIndex = headers.indexOf("approvername") !== -1 ? headers.indexOf("approvername") : 16;
    const rejectionReasonIndex = headers.indexOf("rejectionreason") !== -1 ? headers.indexOf("rejectionreason") : 18;
    
    const documents = data.map(row => ({
      docNumber: row[docNumberIndex] ? row[docNumberIndex].toString().replace(/'/g, "").trim() : "",
      date: (row[dateIndex] instanceof Date) ? row[dateIndex].toISOString() : (row[dateIndex] ? new Date(row[dateIndex]).toISOString() : new Date().toISOString()),
      requester: requesterIndex < row.length ? (row[requesterIndex] || '') : '',
      status: statusIndex < row.length ? (row[statusIndex] || '') : '',
      approverName: approverNameIndex < row.length ? (row[approverNameIndex] || '') : '',
      rejectionReason: rejectionReasonIndex < row.length ? (row[rejectionReasonIndex] || '') : '',
      contactName: contactNameIndex < row.length ? (row[contactNameIndex] || '') : '', 
      companyName: companyNameIndex < row.length ? (row[companyNameIndex] || '') : ''  
    }));

    const pendingCount = documents.filter(doc => doc.status === 'pending').length;

    return { success: true, data: { all: documents, pendingCount: pendingCount } };
  } catch (err) {
    Logger.log(`Error in getDocuments: ${err.stack}`);
    return { success: false, message: `Error fetching documents: ${err.message}` };
  }
}

function getQuotationDetails(docNumber) {
  try {
    if (!docNumber) return { success: false, message: "Document number is required." };
    docNumber = docNumber.toString().replace(/'/g, "").trim();
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    const qSheet = ss.getSheetByName("Quotations");
    const qData = qSheet.getDataRange().getValues();
    if (qData.length < 2) return { success: false, message: `Quotation ${docNumber} not found.` };

    const qHeadersOriginal = qData.shift();
    const qHeadersLower = qHeadersOriginal.map(h => String(h).trim().toLowerCase());
    const qDocNumberIndex = qHeadersLower.indexOf("docnumber") !== -1 ? qHeadersLower.indexOf("docnumber") : 0;
    
    const qRow = qData.find(r => r[qDocNumberIndex] && r[qDocNumberIndex].toString().replace(/'/g, "").trim() === docNumber);

    if (!qRow) return { success: false, message: `Quotation ${docNumber} not found.` };
    
    const defaultHeaders = [
      "docNumber", "date", "requester", "status", "contactName", "contactPhone", "companyName", "address", 
      "deliveryDate", "expiryDate", "creditDays", "machineModel", "technician", 
      "proposerPhone", "grandTotal", "signatureUrl", "approverName", "approverSignatureUrl", 
      "rejectionReason", "remarks", "approvalDate"
    ];

    const quotation = {};
    defaultHeaders.forEach((key, idx) => {
      let val = idx < qRow.length ? qRow[idx] : "";
      const isDateCol = [1, 8, 9, 20].includes(idx);
      if (isDateCol && val instanceof Date) {
        val = val.toISOString();
      }
      quotation[key] = val !== undefined ? val : "";
    });
    quotation.docNumber = docNumber;

    const iSheet = ss.getSheetByName("Items");
    let items = [];
    if (iSheet && iSheet.getLastRow() > 1) {
      const iAll = iSheet.getDataRange().getValues();
      const iHeaders = iAll[0];
      const iHeadersLower = iHeaders.map(h => String(h).trim().toLowerCase());
      const docNumIndexItems = iHeadersLower.indexOf("docnumber") !== -1 ? iHeadersLower.indexOf("docnumber") : 0;

      items = iAll.slice(1)
        .filter(r => r[docNumIndexItems] && r[docNumIndexItems].toString().replace(/'/g, "").trim() === docNumber)
        .map(r => iHeaders.reduce((obj, header, i) => {
          obj[header] = r[i];
          return obj;
        }, {}));
    }
      
    return { success: true, data: { ...quotation, items } };
  } catch(err) {
     Logger.log(`Error in getQuotationDetails: ${err.stack}`);
    return { success: false, message: `Error fetching details: ${err.message}` };
  }
}

function updateQuotationStatus(data) {
  try {
    const { docNumber, status, reason, approverName, signature } = data;
    if (!docNumber || !status) {
      return { success: false, message: "Missing required data for status update." };
    }
    
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Quotations");
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim().toLowerCase());
    const docNumberColIndex = headers.indexOf("docnumber") + 1;

    if (docNumberColIndex === 0) return { success: false, message: "Column docNumber not found." };

    const allDocNumbers = sheet.getRange(2, docNumberColIndex, sheet.getLastRow() - 1, 1).getValues();
    const rowIndex = allDocNumbers.findIndex(row => row[0].toString().replace(/'/g, "") === docNumber);
    
    if (rowIndex === -1) {
      return { success: false, message: `Document ${docNumber} not found for update.` };
    }
    
    const rowToUpdate = rowIndex + 2;
    
    sheet.getRange(rowToUpdate, headers.indexOf("status") + 1).setValue(status);
    const rejectIdx = headers.indexOf("rejectionreason");
    if (rejectIdx !== -1) sheet.getRange(rowToUpdate, rejectIdx + 1).setValue(reason || ""); 
    
    const apprNameIdx = headers.indexOf("approvername");
    const apprSigIdx = headers.indexOf("approversignatureurl");
    const appDateIdx = headers.indexOf("approvaldate");

    if (status === 'approved') {
      let signatureUrl;
      if (signature && signature.startsWith('data:image/png;base64,')) {
          signatureUrl = saveSignatureToDrive(signature, `${docNumber}_approver`);
      } else {
          signatureUrl = signature || "";
      }

      if (apprNameIdx !== -1) sheet.getRange(rowToUpdate, apprNameIdx + 1).setValue(approverName);
      if (apprSigIdx !== -1) sheet.getRange(rowToUpdate, apprSigIdx + 1).setValue(signatureUrl);
      if (appDateIdx !== -1) sheet.getRange(rowToUpdate, appDateIdx + 1).setValue(new Date());
    } else {
      if (apprNameIdx !== -1) sheet.getRange(rowToUpdate, apprNameIdx + 1).setValue("");
      if (apprSigIdx !== -1) sheet.getRange(rowToUpdate, apprSigIdx + 1).setValue("");
      if (appDateIdx !== -1) sheet.getRange(rowToUpdate, appDateIdx + 1).setValue("");
    }
    
    return { success: true, message: `Status updated to ${status}` };
  } catch (err) {
    Logger.log(`Error in updateQuotationStatus: ${err.stack}`);
    return { success: false, message: `Error updating status: ${err.message}` };
  }
}

function generateDocNumber() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName("Quotations");
    const lock = LockService.getScriptLock();
    lock.waitLock(30000); 

    let newDocNumber;
    try {
      const now = new Date();
      const year = now.getFullYear();
      const month = (now.getMonth() + 1).toString().padStart(2, '0');
      const prefix = `QT-${year}${month}-`;

      if (!sheet || sheet.getLastRow() < 2) {
        newDocNumber = `${prefix}001`;
      } else {
        const docNumbers = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues()
                              .flat()
                              .map(d => d.toString().replace(/'/g, ""));
        
        const lastNumberInMonth = docNumbers
          .filter(docNum => docNum.startsWith(prefix))
          .map(docNum => parseInt(docNum.split('-')[2], 10))
          .reduce((max, current) => current > max ? current : max, 0);

        const newNumber = (lastNumberInMonth + 1).toString().padStart(3, '0');
        newDocNumber = `${prefix}${newNumber}`;
      }
    } finally {
      lock.releaseLock();
    }
    
    return { success: true, docNumber: newDocNumber };

  } catch (err) {
    Logger.log(`Error in generateDocNumber: ${err.stack}`);
    return { success: false, message: "Could not generate document number." };
  }
}

      "rejectionReason", "remarks", "approvalDate"
    ];

    const allDocs = [];
    qData.forEach(r => {
      const docNum = r[docNumIdx] ? r[docNumIdx].toString().replace(/'/g, "").trim() : "";
      if (!docNum) return;
      const quotation = {};
      defaultHeaders.forEach((key, idx) => {
        const hIdx = qHeadersLower.indexOf(key.toLowerCase());
        const val = hIdx !== -1 ? r[hIdx] : (idx < r.length ? r[idx] : "");
        if (val instanceof Date) {
          quotation[key] = val.toISOString();
        } else {
          quotation[key] = val !== undefined ? val : "";
        }
      });
      quotation.docNumber = docNum;
      quotation.items = itemsByDocNum[docNum] || [];
      allDocs.push(quotation);
    });

    return { success: true, count: allDocs.length, data: allDocs };
  } catch (err) {
    Logger.log(`Error in getAllQuotationsFull: ${err.stack}`);
    return { success: false, message: `Error fetching all quotations: ${err.message}` };
  }
}
